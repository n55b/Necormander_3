public class UnseenMissHandler : IGemEffectHandler
{
    public GemUniqueType HandledType => GemUniqueType.UnseenMiss;

    public void OnEquipped()
    {
        DamageEventBus.OnEvasionOccurred += OnEvasion;
    }

    public void OnUnequipped()
    {
        DamageEventBus.OnEvasionOccurred -= OnEvasion;
    }

    private void OnEvasion(CharacterHealth target, CharacterStat attackerStat)
    {
        if (attackerStat != null && !attackerStat.IsEnemy && attackerStat.jobType == CommandData.SkeletonArcher)
        {
            var attackerHealth = attackerStat.GetComponent<CharacterHealth>();
            if (attackerHealth == null) return;
            var attackerStatus = attackerHealth.GetComponent<CharacterStatus>();
            
            // 반구저기(ReflectingNature)가 같이 장착되어 있는지 확인
            if (attackerStatus != null && InventoryManager.Instance != null && InventoryManager.Instance.HasUniqueEffect(GemUniqueType.ReflectingNature))
            {
                attackerStatus.LastAttackMissed = true;
            }
        }
    }
}

public class GuillotineHandler : IGemEffectHandler
{
    public GemUniqueType HandledType => GemUniqueType.Guillotine;

    public void OnEquipped()
    {
        DamageEventBus.OnBeforeExecutionCalculated += OnExecution;
    }

    public void OnUnequipped()
    {
        DamageEventBus.OnBeforeExecutionCalculated -= OnExecution;
    }

    private void OnExecution(CharacterHealth target, ref float executeThreshold)
    {
        int count = InventoryManager.Instance.GetUniqueEffectCount(GemUniqueType.Guillotine);
        if (count > 0)
        {
            executeThreshold *= (1.0f + 0.1f * count);
        }
    }
}

Shader "Necromancer/World Atmosphere"
{
    Properties { _Palette ("Artist swatches", 2D) = "white" {} }
    SubShader {
        Tags { "Queue"="Transparent" "RenderType"="Transparent" "RenderPipeline"="UniversalPipeline" "DisableBatching"="True" }
        Pass {
            Tags { "LightMode"="Universal2D" }
            // The framebuffer already contains the sharp artwork. Add only scattered light and mist.
            Blend One OneMinusSrcAlpha, Zero One
            ZWrite Off Cull Off ZTest Always
            HLSLPROGRAM
            #pragma vertex Vert
            #pragma fragment Frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"
            TEXTURE2D(_CameraSortingLayerTexture); TEXTURE2D(_Palette);
            float _Mist, _Bloom;
            struct A { float3 pos:POSITION; };
            struct V { float4 pos:SV_POSITION; float2 world:TEXCOORD0; };
            V Vert(A a) {
                V o;
                float2 clipXY = a.pos.xy * 2;
                float3 world = mul(UNITY_MATRIX_I_V, float4(clipXY * unity_OrthoParams.xy, -3, 1)).xyz;
                o.pos = TransformWorldToHClip(world);
                o.world = world.xy;
                return o;
            }
            float Hash(float2 p) {
                float3 q = frac(p.xyx * .1031);
                q += dot(q, q.yzx + 33.33); return frac((q.x+q.y)*q.z);
            }
            float Noise(float2 p) {
                float2 i=floor(p), f=frac(p); f=f*f*(3-2*f);
                return lerp(lerp(Hash(i),Hash(i+float2(1,0)),f.x),
                            lerp(Hash(i+float2(0,1)),Hash(i+1),f.x),f.y);
            }
            half3 Swatch(float col,float row) {
                return SAMPLE_TEXTURE2D(_Palette,sampler_PointClamp,float2((col+.5)/16,1-(row+.5)/9)).rgb;
            }
            half3 Bright(float2 uv) {
                half3 c = SAMPLE_TEXTURE2D(_CameraSortingLayerTexture,sampler_PointClamp,uv).rgb;
                return c * smoothstep(.58, .98, max(c.r,max(c.g,c.b)));
            }
            half4 Frag(V i):SV_Target {
                float2 uv = GetNormalizedScreenSpaceUV(i.pos);
                float2 px = rcp(_ScaledScreenParams.xy);
                half3 glow = 0;
                [unroll] for (int k=0;k<4;k++) {
                    float2 d = k==0 ? float2(1,0) : (k==1 ? float2(-1,0) : (k==2 ? float2(0,1) : float2(0,-1)));
                    glow += .15*Bright(uv+2*d*px) + .1*Bright(uv+5*float2(d.x-d.y,d.y+d.x)*px);
                }
                float2 p = floor(i.world * 32) / 32;
                float t = _Time.y;
                float cloud = .70*Noise(p*float2(.24,.52)+float2(-t*.066,t*.015))+
                              .30*Noise(p*float2(.68,1.1)-float2(t*.09,0));
                float shaft = smoothstep(.55,.92, Noise(float2(p.x*.55-p.y*.31-t*.024, 11.3)));
                float haze = saturate((cloud-.27)*.85+shaft*.17);
                float grain = Hash(floor(p*32));
                float alpha = floor(haze*16+grain)/16 * _Mist;
                half3 tint = lerp(Swatch(10,7),Swatch(1,7),saturate(cloud*.88+shaft*.25));
                return half4(tint*alpha + glow*_Bloom*.48, alpha);
            }
            ENDHLSL
        }
    }
}

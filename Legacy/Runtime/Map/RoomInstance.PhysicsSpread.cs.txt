    public void CleanupPhysics()
    {
        if (_rb != null)
        {
            _rb.simulated = false;
            _rb.bodyType = RigidbodyType2D.Static;
        }
        if (_physicsCollider != null) MapGenerator.SafeDestroy(_physicsCollider);
        if (_rb != null) MapGenerator.SafeDestroy(_rb);
    }
